53class AnnouncementQuerySet(models.QuerySet): 

54 def live(self, now: datetime.datetime | None = None): 

55 """``starts_at`` ≤ ``now`` < ``ends_at`` の行。""" 

56 now = now or timezone.now() 

57 return self.filter(starts_at__lte=now, ends_at__gt=now) 

58 

59 def scheduled(self, now: datetime.datetime | None = None): 

60 return self.filter(starts_at__gt=now or timezone.now()) 

61 

62 def ended(self, now: datetime.datetime | None = None): 

63 return self.filter(ends_at__lte=now or timezone.now())

65 def for_extension_version(self, version: str): 

66 """``version``（``x.y.z``）が配信対象の行。下限・上限はどちらも含み、空の側は制限なし。 

67 

68 比較は DB の整数（``versions.encode``）で行うので、``2.10.0`` > ``2.9.0`` になる。 

69 形の合わない ``version`` は ``ValueError``。 

70 """ 

71 return self.filter( 

72 Q(min_extension_version__isnull=True) 

73 | Q(min_extension_version__lte=version), 

74 Q(max_extension_version__isnull=True) 

75 | Q(max_extension_version__gte=version), 

76 )

140 def __str__(self) -> str:

143 def clean(self) -> None: 

144 super().clean() 

145 if self.starts_at and self.ends_at and self.starts_at >= self.ends_at: 

146 raise ValidationError({"ends_at": ENDS_BEFORE_STARTS}) 

147 # 形の合わない値は clean_fields がその項目のエラーにするので、ここでは比べない。 

148 lower = versions.parse(self.min_extension_version) 

149 upper = versions.parse(self.max_extension_version) 

150 if lower and upper and lower > upper: 

151 raise ValidationError({"max_extension_version": MAX_VERSION_BELOW_MIN})

153 def target_versions(self) -> str: 

154 """配信対象のバージョンの範囲（例: ``2.15.0 〜 2.16.3``・``〜 2.14.1``・``すべて``）。""" 

155 lower, upper = self.min_extension_version, self.max_extension_version 

156 if lower is None and upper is None: 

157 return "すべて" 

158 return " ".join(part for part in (lower, "〜", upper) if part)

160 def status(self, now: datetime.datetime | None = None) -> AnnouncementStatus: 

161 now = now or timezone.now() 

162 if now < self.starts_at: 

163 return AnnouncementStatus.SCHEDULED 

164 if now < self.ends_at: 

165 return AnnouncementStatus.LIVE 

166 return AnnouncementStatus.ENDED
