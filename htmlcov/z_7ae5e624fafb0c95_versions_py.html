1"""拡張機能のバージョン（``x.y.z``）の解釈と、DB に整数で持つためのモデルのフィールド。 

2 

3アナウンスの配信対象（下限・上限）を DB で絞り込むため、バージョンを「数値の順に並ぶ整数」に 

4直して 1 列で持つ。各部を 10 進 5 桁ずつに詰める（``2.15.0`` → ``2_00015_00000``）。 

5Chrome の manifest の各部は 0〜65535 なので 5 桁に収まり、整数の大小がそのままバージョンの 

6大小になる（``2.10.0`` > ``2.9.0``）。Python 側では ``x.y.z`` の文字列で扱う。 

7""" 

8 

9import re 

10from typing import Any 

11 

12from django import forms 

13from django.core.exceptions import ValidationError 

14from django.db import models 

15 

16# Chrome の manifest の version と同じ規則（ただし数字 3 つに限る）。各部は 0〜65535 の整数で、 

17# 0 以外は 0 で始めない。OpenAPI の pattern にも使うので ECMAScript でも読める形にする。 

18PATTERN = r"^(0|[1-9][0-9]{0,4})\.(0|[1-9][0-9]{0,4})\.(0|[1-9][0-9]{0,4})$" 

19COMPONENT_MAX = 65535 

20# ``65535.65535.65535`` の長さ。 

21MAX_LENGTH = 17 

22 

23INVALID_VERSION = ( 

24 "x.y.z の形（例: 2.15.0）で入力してください。各部は 0〜65535 の整数です。" 

25) 

26 

27_VERSION_RE = re.compile(PATTERN) 

28_BASE = 100_000 

29 

30 

31def parse(value: object) -> tuple[int, int, int] | None: 

32 """``x.y.z`` を ``(x, y, z)`` に直す。形が合わなければ ``None``。""" 

33 # ``$`` は末尾の改行の前でも当たるので、fullmatch で文字列全体を見る。 

34 if not isinstance(value, str) or not (match := _VERSION_RE.fullmatch(value)): 

35 return None 

36 major, minor, patch = (int(part) for part in match.groups()) 

37 if max(major, minor, patch) > COMPONENT_MAX: 

38 return None 

39 return major, minor, patch 

40 

41 

42def encode(value: str) -> int: 

43 """``x.y.z`` を数値の順に並ぶ整数に直す。形が合わなければ ``ValueError``。""" 

44 parts = parse(value) 

45 if parts is None: 

46 raise ValueError(f"not an x.y.z extension version: {value!r}") 

47 major, minor, patch = parts 

48 return (major * _BASE + minor) * _BASE + patch 

49 

50 

51def decode(number: int) -> str: 

52 """``encode`` の逆。""" 

53 rest, patch = divmod(number, _BASE) 

54 major, minor = divmod(rest, _BASE) 

55 return f"{major}.{minor}.{patch}" 

56

58class ExtensionVersionField(models.Field): 

59 """``x.y.z`` の文字列を ``bigint`` の 1 列で持つ。 

60 

61 比較の lookup（``__lte`` · ``__gte`` など）に文字列を渡すと整数に直して DB で比べる。 

62 """ 

63 

64 description = "拡張機能のバージョン（x.y.z）" 

65 

66 def get_internal_type(self) -> str: 

67 return "BigIntegerField" 

68 

69 def from_db_value(self, value: int | None, expression: Any, connection: Any): 

70 return None if value is None else decode(value) 

71 

72 def to_python(self, value: Any) -> str | None: 

73 if value is None or value == "": 

74 return None 

75 if isinstance(value, int): 

76 return decode(value) 

77 if parse(value) is None: 

78 raise ValidationError(INVALID_VERSION, code="invalid") 

79 return value 

80 

81 def get_prep_value(self, value: Any) -> int | None: 

82 value = super().get_prep_value(value) 

83 # 空文字は「制限なし」（to_python と同じ扱い）。 

84 return None if value in (None, "") else encode(value)

87 defaults: dict[str, Any] = { 

88 "form_class": forms.CharField,

91 if self.null:

93 defaults["empty_value"] = None
