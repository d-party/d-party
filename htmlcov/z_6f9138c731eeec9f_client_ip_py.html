1"""リクエストを送ってきたクライアントの IP。django-axes のロックに使う。 

2 

3本番は Cloudflare → cloudflared → nginx → Django とつながるので、``REMOTE_ADDR`` は 

4nginx の Pod になる。クライアントの IP は Cloudflare が付ける ``CF-Connecting-IP`` で取る。 

5Cloudflare はクライアントが送った同名のヘッダーを上書きするので、Cloudflare を通る限り 

6偽れない。nginx は cloudflared の後ろの ClusterIP で、クラスタの外からは直接届かない 

7（クラスタ内の別の Pod からは届くので、同居するサービスは信頼している前提）。 

8""" 

9 

10import ipaddress 

11import logging 

12from typing import TYPE_CHECKING 

13 

14from django.conf import settings 

15 

16if TYPE_CHECKING: 

17 from django.http import HttpRequest 

18 

19logger = logging.getLogger(__name__) 

20 

21# IPv6 は 1 契約者が /56〜/48 を持つので、アドレスごとに数えると、その中で回すだけで 

22# 試行回数の上限を回避できる。/56 でまとめて数える。 

23IPV6_PREFIX_LENGTH = 56 

24 

25 

26def _lockout_key(address: str) -> str: 

27 """ロックを数える単位。IPv4 はそのまま、IPv6 は /56 のネットワーク。""" 

28 ip = ipaddress.ip_address(address) 

29 if isinstance(ip, ipaddress.IPv6Address): 

30 if ip.ipv4_mapped: 

31 return str(ip.ipv4_mapped) 

32 network = ipaddress.IPv6Network((ip, IPV6_PREFIX_LENGTH), strict=False) 

33 return str(network.network_address) 

34 return str(ip) 

35 

36 

37def client_ip(request: HttpRequest) -> str | None: 

38 """``CF-Connecting-IP`` が IP として読めればその単位、なければ ``REMOTE_ADDR``。""" 

39 forwarded = request.META.get("HTTP_CF_CONNECTING_IP", "").strip() 

40 if forwarded: 

41 try: 

42 return _lockout_key(forwarded) 

43 except ValueError: 

44 pass 

45 if not settings.DEBUG: 

46 # 本番で REMOTE_ADDR（nginx の Pod）に戻ると、全員が同じ IP として数えられ、 

47 # ユーザー名だけのロックと同じになる。黙って起きないように残す。 

48 logger.warning("no usable CF-Connecting-IP; falling back to REMOTE_ADDR") 

49 return request.META.get("REMOTE_ADDR")
