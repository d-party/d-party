1"""アナウンスの本文に書けるリンク（Markdown のインラインリンク ``[テキスト](URL)``）。 

2 

3拡張機能（``extension/src/domain/announcementText.ts``）と同じ規則で読む。片側だけ変えないこと。 

4 

5- 書けるのはインラインリンクだけ。ほかの Markdown（太字・見出し・自動リンクなど）は扱わず、 

6 そのまま文字として出る。1 つの本文に複数のリンクを書いてよい。 

7- リンクにできる URL は ``https://`` で始まり、ホストがあり、空白・制御文字・バックスラッシュ・ 

8 ユーザー情報（``https://user@host``）を含まないものだけ。 

9- 本文の長さは「表示される文字数」（リンクはテキストだけを数え、URL は数えない）と、URL を 

10 含む生の文字列の長さの 2 つで制限する。 

11 

12管理画面では、許可しない URL のリンク・空のテキスト・崩れた書き方をフォームのエラーにする。 

13拡張機能は DB に直接入った値にも備えて同じ検査をし、満たさないものは元の文字列のまま出す。 

14""" 

15 

16import re 

17import unicodedata 

18from dataclasses import dataclass 

19from urllib.parse import urlsplit 

20 

21from django.core.exceptions import ValidationError 

22 

23# 表示される文字数の上限。拡張機能のポップアップ（幅 360px）とサイドバー（幅 320px）に 

24# 2〜3 行で収まる長さ。 

25DISPLAY_MAX_LENGTH = 60 

26# URL を含む生の文字列の上限（DB の列の長さ）。 

27RAW_MAX_LENGTH = 500 

28 

29# ``[テキスト](URL)``。各グループは否定の文字クラス 1 つだけで、その直後の区切り文字 

30# （``]`` · ``)``）をクラスに含まないので、バックトラックが積み重ならない（ReDoS にならない）。 

31# テキストに ``[ ]``、URL に ``( )`` は書けない（URL の括弧は ``%28`` ``%29`` にする）。 

32LINK_RE = re.compile(r"\[([^\[\]]*)\]\(([^()]*)\)") 

33 

34# リンクの書き方の名残り。正しいリンクを取り除いた後に残っていれば、崩れた書き方とみなす。 

35_BROKEN_LINK_MARK = "](" 

36 

37ALLOWED_SCHEME_PREFIX = "https://" 

38 

39 

40@dataclass(frozen=True) 

41class Link: 

42 text: str 

43 url: str 

44 

45 

46type Segment = str | Link 

47 

48 

49def is_allowed_url(url: str) -> bool: 

50 """リンクにしてよい URL か（``https://`` で始まり、ホストがあり、空白・制御文字・ 

51 バックスラッシュ・ユーザー情報を含まない）。""" 

52 if not url.startswith(ALLOWED_SCHEME_PREFIX): 

53 return False 

54 # バックスラッシュは、ブラウザ（WHATWG URL）が / と読み、Python の urlsplit と食い違うので弾く。 

55 if "\\" in url or any( 

56 char.isspace() or unicodedata.category(char)[0] == "C" for char in url 

57 ): 

58 return False 

59 try: 

60 parts = urlsplit(url) 

61 hostname = parts.hostname 

62 except ValueError: # 角括弧の閉じない IPv6 など 

63 return False 

64 return ( 

65 parts.scheme == "https" 

66 and bool(hostname) 

67 and parts.username is None 

68 and parts.password is None 

69 )

72def _as_link(text: str, url: str) -> Link | None: 

73 if not text.strip() or not is_allowed_url(url): 

74 return None 

75 return Link(text, url)

78def parse(message: str) -> list[Segment]: 

79 """本文を文字列とリンクの並びに分ける。リンクにできないものは元の文字列のまま残す。""" 

80 segments: list[Segment] = [] 

81 rest = 0 

82 for match in LINK_RE.finditer(message): 

83 link = _as_link(match[1], match[2]) 

84 if link is None: 

85 continue 

86 if match.start() > rest: 

87 segments.append(message[rest : match.start()]) 

88 segments.append(link) 

89 rest = match.end() 

90 if rest < len(message): 

91 segments.append(message[rest:]) 

92 return segments

95def display_text(message: str) -> str: 

96 """拡張機能に表示される文字列（リンクはテキストだけにする）。""" 

97 return "".join( 

98 segment.text if isinstance(segment, Link) else segment 

99 for segment in parse(message) 

100 )

103def display_length(message: str) -> int: 

104 """表示される文字数。リンクはテキストを数え、URL は数えない。""" 

105 return len(display_text(message))

108def link_errors(message: str) -> list[str]: 

109 """リンクの書き方の誤り（許可しない URL · 空のテキスト · 崩れた書き方）。""" 

110 errors: list[str] = [] 

111 for match in LINK_RE.finditer(message): 

112 text, url = match[1], match[2] 

113 if not text.strip(): 

114 errors.append(f"「{match[0]}」はリンクのテキストが空です。") 

115 elif not is_allowed_url(url): 

116 errors.append( 

117 f"「{match[0]}」の URL はリンクにできません。" 

118 "https:// で始まり、空白を含まない URL だけを使えます。" 

119 ) 

120 if _BROKEN_LINK_MARK in LINK_RE.sub("", message): 

121 errors.append( 

122 "リンクの書き方が崩れています。[テキスト](https://…) の形で書いてください" 

123 "（テキストに [ ]、URL に ( ) は使えません。URL の括弧は %28 %29 にします）。" 

124 ) 

125 return errors

128def validate_message(message: str) -> None: 

129 """本文の検査（モデルのフィールドの validator）。生の文字列の長さは ``max_length`` が見る。""" 

130 errors = link_errors(message) 

131 # 書き方に誤りがあると、そのリンクは URL ごと数えてしまうので、文字数は直した後に見る。 

132 length = display_length(message) 

133 if not errors and length > DISPLAY_MAX_LENGTH: 

134 errors.append( 

135 f"表示される文字数は {DISPLAY_MAX_LENGTH} 文字までです（今は {length} 文字。" 

136 "リンクはテキストだけを数え、URL は数えません）。" 

137 ) 

138 if errors: 

139 raise ValidationError(errors, code="invalid_announcement_message")
